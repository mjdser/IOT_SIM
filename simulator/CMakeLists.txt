cmake_minimum_required(VERSION 3.16)
project(lvgl_bench_host LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS vendor/lvgl/src/*.c)
add_executable(lvgl_sim host/sim_main.cpp ../lvgl_ui/generated_ui.cpp ../lvgl_ui/test_hooks.cpp ${LVGL_SOURCES})
target_compile_definitions(lvgl_sim PRIVATE TEST_HOOKS LV_CONF_INCLUDE_SIMPLE)
target_include_directories(lvgl_sim PRIVATE host ../include ../lvgl_ui vendor/lvgl)
find_package(Threads REQUIRED)
target_link_libraries(lvgl_sim PRIVATE Threads::Threads)
if(NOT WIN32)
  target_link_libraries(lvgl_sim PRIVATE m)
endif()
